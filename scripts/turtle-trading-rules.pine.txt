// The Turtle Trading rules (Richard Dennis & William Eckhardt, 1983) as published by former Turtles:
// enter on a 20-day (System 1) or 55-day (System 2) breakout, size each unit so 1 N (20-day ATR) = 1% of
// equity, stop 2 N from the last unit, add a unit every ½ N up to 4, exit on a 10-day / 20-day opposite breakout.
// Defaults = the setting optimised on gold daily 2006-2018 and validated on 2019-2026 (55/20 days, 1.5 N stop, 4 units, long only);
// the textbook System 1 is 20/10 days, 2 N stop, long and short.
strategy("Turtle Breakout (Dennis & Eckhardt) [JR]", shorttitle = "Turtles", overlay = true,
     initial_capital = 100000, commission_type = strategy.commission.percent, commission_value = 0.02, slippage = 2,
     margin_long = 0, margin_short = 0, pyramiding = 4, process_orders_on_close = false)

G1 = "1 · Breakout"
entryLen = input.int(55, "Entry: breakout of the last N days", minval = 5, group = G1, tooltip = "System 1 = 20, System 2 = 55")
exitLen  = input.int(20, "Exit: opposite breakout of the last N days", minval = 2, group = G1, tooltip = "System 1 = 10, System 2 = 20")
skipWin  = input.bool(false, "System 1 filter: skip the next breakout after a winner", group = G1)
dirMode  = input.string("Long only", "Direction", options = ["Both", "Long only", "Short only"], group = G1)
G2 = "2 · N, stops and units"
nLen     = input.int(20, "N = ATR length", minval = 5, group = G2)
stopN    = input.float(1.5, "Stop: last unit ± N ×", minval = 0.5, step = 0.25, group = G2)
riskPct  = input.float(1.0, "Unit size: 1 N move = % of equity", minval = 0.1, step = 0.1, group = G2)
maxUnits = input.int(4, "Max units", minval = 1, maxval = 4, group = G2)
addN     = input.float(0.5, "Add a unit every N ×", minval = 0.25, step = 0.25, group = G2)
maxLev   = input.float(2.0, "Cap: one unit ≤ equity × (leverage)", minval = 0.1, step = 0.1, group = G2)
G3 = "3 · Test window"
fromT  = input.time(timestamp("2000-01-01T00:00:00"), "Trade from", group = G3)
toT    = input.time(timestamp("2100-01-01T00:00:00"), "Trade until", group = G3)
showLv = input.bool(true, "Show channels", group = G3)

inWin   = time >= fromT and time <= toT
N       = ta.atr(nLen)
hiEntry = ta.highest(high, entryLen)
loEntry = ta.lowest(low, entryLen)
hiExit  = ta.highest(high, exitLen)
loExit  = ta.lowest(low, exitLen)
ready   = bar_index > math.max(entryLen, nLen) + 1 and not na(N) and not na(hiEntry) and not na(loEntry)   // a stop at na would be a market order
canLong = dirMode != "Short only"
canShrt = dirMode != "Long only"

unitQty = math.min(strategy.equity * riskPct / 100 / (N * syminfo.pointvalue), strategy.equity * maxLev / 4 / (close * syminfo.pointvalue))

var float nEntry = na
var bool  lastWin = false
if strategy.position_size != 0 and strategy.position_size[1] == 0
    nEntry := na(N[1]) ? N : N[1]
if strategy.closedtrades > strategy.closedtrades[1]
    lastWin := strategy.closedtrades.profit(strategy.closedtrades - 1) > 0

strategy.cancel_all()
units = strategy.opentrades
if strategy.position_size == 0 and inWin and ready and not (skipWin and lastWin)
    if canLong
        strategy.entry("T L1", strategy.long, qty = unitQty, stop = hiEntry + syminfo.mintick)
    if canShrt
        strategy.entry("T S1", strategy.short, qty = unitQty, stop = loEntry - syminfo.mintick)
if strategy.position_size == 0 and skipWin and lastWin and (high > hiEntry[1] or low < loEntry[1])
    lastWin := false   // the skipped breakout happened; the next one is taken

var float stopPx = na
if strategy.position_size > 0
    lastP = strategy.opentrades.entry_price(units - 1)
    stopPx := math.max(lastP - stopN * nEntry, loExit)
    if units < maxUnits and inWin and not na(nEntry)
        strategy.entry("T L" + str.tostring(units + 1), strategy.long, qty = unitQty, stop = lastP + addN * nEntry)
    strategy.exit("T L out", stop = stopPx)
else if strategy.position_size < 0
    lastP = strategy.opentrades.entry_price(units - 1)
    stopPx := math.min(lastP + stopN * nEntry, hiExit)
    if units < maxUnits and inWin and not na(nEntry)
        strategy.entry("T S" + str.tostring(units + 1), strategy.short, qty = unitQty, stop = lastP - addN * nEntry)
    strategy.exit("T S out", stop = stopPx)
else
    stopPx := na
if not inWin and strategy.position_size != 0
    strategy.close_all(comment = "window end")

plot(showLv ? hiEntry : na, "Entry high (next bar)", color.new(color.teal, 30), 1, offset = 1)
plot(showLv ? loEntry : na, "Entry low (next bar)", color.new(color.red, 30), 1, offset = 1)
plot(strategy.position_size != 0 ? stopPx : na, "Stop / exit", color.new(color.orange, 0), 2, plot.style_linebr, offset = 1)
